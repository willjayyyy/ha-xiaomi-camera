# Pinned to a commit rather than a branch. This package brings in a
# closed-source native library that is loaded into a process on the host

# upstream changes to the bundled library when you do.

# numpy is an optional dependency of the SDK decoder; without it every
# decoded frame logs a conversion warning.
numpy>=1.26

# aiohttp arrives transitively via the SDK, but the bridge's HTTP control plane
# and ingress UI depend on it directly, so it is declared here as well.
aiohttp>=3.14.1

# PyAV arrives transitively via the SDK, whose decoder is built on it, but the
# bridge now packages the published stream with it directly -- so it is
# declared here as well. Pinned to a major version: `add_mux_stream` is the
# whole reason it is used, and it arrived in 14.
av>=14
